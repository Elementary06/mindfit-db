CREATE TYPE "tip_client_enum" AS ENUM (
  'persoana_fizica',
  'firma'
);

CREATE TYPE "tip_adresa_enum" AS ENUM (
  'livrare',
  'facturare'
);

CREATE TYPE "status_comanda_enum" AS ENUM (
  'noua',
  'in_productie',
  'gata',
  'livrata',
  'anulata'
);

CREATE TYPE "status_plata_enum" AS ENUM (
  'neplatita',
  'partial',
  'platita',
  'rambursata'
);

CREATE TYPE "metoda_plata_enum" AS ENUM (
  'card',
  'ramburs',
  'transfer',
  'numerar'
);

CREATE TYPE "rol_angajat_enum" AS ENUM (
  'designer',
  'operator_print',
  'vanzari',
  'administrator'
);

CREATE TABLE "clienti" (
  "id_client" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "prenume" varchar(50) NOT NULL,
  "nume" varchar(50) NOT NULL,
  "email" varchar(100) UNIQUE NOT NULL,
  "telefon" varchar(20),
  "tip_client" tip_client_enum NOT NULL DEFAULT 'persoana_fizica',
  "denumire_firma" varchar(100),
  "cui" varchar(20),
  "data_creare" timestamp NOT NULL DEFAULT (now())
);

CREATE TABLE "adrese" (
  "id_adresa" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "id_client" integer NOT NULL,
  "tip_adresa" tip_adresa_enum NOT NULL,
  "strada" varchar(150) NOT NULL,
  "oras" varchar(60) NOT NULL,
  "judet" varchar(60),
  "cod_postal" varchar(10),
  "tara" varchar(50) NOT NULL DEFAULT 'Romania'
);

CREATE TABLE "categorii" (
  "id_categorie" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "nume_categorie" varchar(60) UNIQUE NOT NULL,
  "descriere" text
);

CREATE TABLE "produse" (
  "id_produs" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "id_categorie" integer NOT NULL,
  "nume_produs" varchar(100) NOT NULL,
  "descriere" text,
  "marca" varchar(60),
  "material" varchar(60),
  "pret_baza" numeric(10,2) NOT NULL,
  "este_activ" boolean NOT NULL DEFAULT true
);

CREATE TABLE "variante_produs" (
  "id_varianta" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "id_produs" integer NOT NULL,
  "marime" varchar(10) NOT NULL,
  "culoare" varchar(30) NOT NULL,
  "cod_sku" varchar(40) UNIQUE NOT NULL,
  "stoc" integer NOT NULL DEFAULT 0,
  "pret_suplimentar" numeric(10,2) NOT NULL DEFAULT 0
);

CREATE TABLE "tehnici_print" (
  "id_tehnica" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "nume_tehnica" varchar(50) UNIQUE NOT NULL,
  "descriere" text,
  "pret_per_bucata" numeric(10,2) NOT NULL,
  "cantitate_minima" integer NOT NULL DEFAULT 1
);

CREATE TABLE "pozitii_print" (
  "id_pozitie" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "nume_pozitie" varchar(50) UNIQUE NOT NULL,
  "latime_max_cm" numeric(5,1),
  "inaltime_max_cm" numeric(5,1),
  "pret_suplimentar" numeric(10,2) NOT NULL DEFAULT 0
);

CREATE TABLE "modele_grafice" (
  "id_model" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "id_client" integer NOT NULL,
  "nume_model" varchar(100) NOT NULL,
  "cale_fisier" varchar(255),
  "tip_fisier" varchar(10),
  "text_personalizat" varchar(200),
  "font" varchar(50),
  "culoare_text" varchar(30),
  "aprobat" boolean NOT NULL DEFAULT false,
  "data_incarcare" timestamp NOT NULL DEFAULT (now())
);

CREATE TABLE "angajati" (
  "id_angajat" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "prenume" varchar(50) NOT NULL,
  "nume" varchar(50) NOT NULL,
  "rol" rol_angajat_enum NOT NULL,
  "email" varchar(100) UNIQUE,
  "telefon" varchar(20),
  "data_angajare" date
);

CREATE TABLE "comenzi" (
  "id_comanda" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "id_client" integer NOT NULL,
  "id_adresa_livrare" integer,
  "id_adresa_facturare" integer,
  "id_angajat" integer,
  "data_comanda" timestamp NOT NULL DEFAULT (now()),
  "data_limita" date,
  "status" status_comanda_enum NOT NULL DEFAULT 'noua',
  "status_plata" status_plata_enum NOT NULL DEFAULT 'neplatita',
  "metoda_livrare" varchar(50),
  "cost_livrare" numeric(10,2) NOT NULL DEFAULT 0,
  "suma_totala" numeric(10,2) NOT NULL DEFAULT 0,
  "observatii" text
);

CREATE TABLE "articole_comanda" (
  "id_articol" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "id_comanda" integer NOT NULL,
  "id_varianta" integer NOT NULL,
  "cantitate" integer NOT NULL,
  "pret_unitar" numeric(10,2) NOT NULL
);

CREATE TABLE "personalizari_articol" (
  "id_personalizare" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "id_articol" integer NOT NULL,
  "id_model" integer NOT NULL,
  "id_tehnica" integer NOT NULL,
  "id_pozitie" integer NOT NULL,
  "pret_personalizare" numeric(10,2) NOT NULL DEFAULT 0,
  "previzualizare_aprobata" boolean NOT NULL DEFAULT false
);

CREATE TABLE "plati" (
  "id_plata" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "id_comanda" integer NOT NULL,
  "data_plata" timestamp NOT NULL DEFAULT (now()),
  "suma" numeric(10,2) NOT NULL,
  "metoda_plata" metoda_plata_enum NOT NULL,
  "referinta_tranzactie" varchar(100),
  "status" varchar(30)
);

CREATE TABLE "facturi" (
  "id_factura" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "id_comanda" integer UNIQUE NOT NULL,
  "numar_factura" varchar(30) UNIQUE NOT NULL,
  "data_emitere" date NOT NULL,
  "suma_totala" numeric(10,2) NOT NULL,
  "suma_tva" numeric(10,2) NOT NULL
);

CREATE TABLE "furnizori" (
  "id_furnizor" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "denumire_furnizor" varchar(100) NOT NULL,
  "persoana_contact" varchar(100),
  "email" varchar(100),
  "telefon" varchar(20),
  "oras" varchar(60)
);

CREATE TABLE "produse_furnizori" (
  "id_furnizor" integer NOT NULL,
  "id_produs" integer NOT NULL,
  "pret_achizitie" numeric(10,2) NOT NULL,
  "termen_livrare_zile" integer,
  PRIMARY KEY ("id_furnizor", "id_produs")
);

COMMENT ON COLUMN "pozitii_print"."nume_pozitie" IS 'piept stanga, piept centru, spate, maneca';

COMMENT ON COLUMN "modele_grafice"."cale_fisier" IS 'cale sau URL catre fisierul grafic';

COMMENT ON COLUMN "modele_grafice"."tip_fisier" IS 'PNG, SVG, PDF';

COMMENT ON COLUMN "articole_comanda"."cantitate" IS 'trebuie sa fie > 0';
